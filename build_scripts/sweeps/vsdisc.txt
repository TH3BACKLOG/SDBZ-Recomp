# vsdisc.txt - DISCOVERY run for the Versus select screen: RAM snapshots at known cursor positions so
# build_scripts\ramdiff.py can name the P1 / P2 / stage cursor words; also a first test of the new ops
# (snap, freeze, fuzz). Short: ~15 min. Needs the build that has `snap`/`freeze`/`fuzz` (10-07 evening).
# Run:  $env:PS2X_PAD2='1'
#       .\build_scripts\gfx_tour.ps1 -Name vsdisc -Script build_scripts\sweeps\vsdisc.txt -Seconds 900 -Every 90
# Then (snapshots land in gsdump\vsdisc\snap\, 32 MB each - copy what you need before the next gfx_tour run,
# it wipes the folder):
#   python build_scripts\ramdiff.py gsdump\vsdisc\snap --steps a0 a1 a2 a3 --noise a0 a0n     (P1 cursor)
#   python build_scripts\ramdiff.py gsdump\vsdisc\snap --steps b0 b1 b2 b3 --noise b0 b0n     (P2 cursor)
#   python build_scripts\ramdiff.py gsdump\vsdisc\snap --steps c0 c1 c2 c3 --noise c0 c0n     (stage cursor)
# Menu path = versus2.txt (VERIFIED vs1). Cursor moves = R (pad 1 for P1, pad 2 for P2, pad 1 for stage).
# H: P1 cursor byte 0x5B4070 already tracks the P1 fighter (Goku 1, Krillin 4); a0..a3 should rediscover it.
# VERIFIED static: fighter array 0x5ADFB0, stride 0x1CD0; fighter1 health ptr 0x5B14C0 -> fighter0 = 0x5AF7F0;
#   health int at obj+12, 250 at fight start (vs1 log 'old=250').

wait app=CAppWarning timeout=900
press X hold=10 gap=50 until app!=CAppWarning timeout=3000
label boot
press S hold=10 gap=50 until app=CAppMenuMain timeout=1500
goto boot 6 if app!=CAppMenuMain
poke 0x5E6B30 0xFFF0 w=4
poke 0x5EC2F0 0x7FE w=4
poke 0x5C7942 0
poke 0x5C7943 0
poke 0x5C7944 0
log unlock + 1 round 45s + stage select on
wait ticks=120

label menu
wait app=CAppMenuMain timeout=3000
wait ticks=90
repeat 4
  press U hold=6 gap=6
  press L hold=6 gap=6
end
press D hold=6 gap=30
press R hold=6 gap=30
log menu path D R (expect mode=3)
press X hold=10 gap=60 until app=CAppSelectMain timeout=600
goto vs_ok if mode=3
log NOT VERSUS - backing out, trying pad 2 Start first
press T hold=10 gap=60 until app=CAppMenuMain timeout=300
press S hold=10 gap=60 pad=2
goto menu 2
label vs_ok
log in Versus char select

# --- P1 cursor: a0 = start, a0n = same screen 60 ticks later (noise), then 3 moves right
wait ticks=200
snap a0
wait ticks=60
snap a0n
press R hold=6 gap=30
wait ticks=40
snap a1
press R hold=6 gap=30
wait ticks=40
snap a2
press R hold=6 gap=30
wait ticks=40
snap a3
press X hold=10 gap=60
log P1 confirmed

# --- P2 cursor (pad 2)
wait ticks=200
snap b0
wait ticks=60
snap b0n
press R hold=6 gap=30 pad=2
wait ticks=40
snap b1
press R hold=6 gap=30 pad=2
wait ticks=40
snap b2
press R hold=6 gap=30 pad=2
wait ticks=40
snap b3
press X hold=10 gap=90 pad=2
log P2 confirmed
wait ticks=120

# --- handicap panel -> stage panel
press X hold=10 gap=120
log handicap done, stage panel
wait ticks=100
snap c0
wait ticks=60
snap c0n
press R hold=6 gap=60
wait ticks=40
snap c1
press R hold=6 gap=60
wait ticks=40
snap c2
press R hold=6 gap=60
wait ticks=40
snap c3
press X hold=10 gap=120 until app=CAppFightMain timeout=900
press X hold=10 gap=120 pad=2 until app=CAppFightMain timeout=600

# --- in the fight: baseline, then freeze both health bars at 250 and fuzz 900 ticks (test of the new ops)
wait ticks=300
snap f0
log fuzz test: health frozen at 250
freeze [0x5AF7F0]+12 250 w=4
freeze [0x5B14C0]+12 250 w=4
fuzz ticks=900 seed=1 max=2 hold=2-8 gap=2-8
snap f1
unfreeze
log fuzz test done
wait ticks=300
