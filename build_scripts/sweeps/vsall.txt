# vsall.txt - walk-away Versus sweep: a new P1 / P2 fighter and stage each loop (one R each, H: cursors
# remember their place), health frozen while BOTH pads mash a seeded fuzz (random chords -> specials/supers,
# H), then health released so the fight ends, result menu D X -> back to the select screen.
# VERIFIED (10-07): menu path D R -> mode 3; stage panel needs X on BOTH pads (pad 2 X starts the fight);
#   fight health ptrs [0x5AF7F0]/[0x5B14C0]+12 (250); fuzz/freeze/snap ops run. Char-id words 0x5B4070 etc are
#   mirrors: freezing them did NOT pick the fighter (vspoke: ids came out 1,2 not 5,9) -> R-stepping instead.
# H: result handling (versus2 'label result') works - versus2 was never run to a result menu.
# Watch the log: [scene] p1=0x.. per loop shows whether P1 changes (cursor memory), [padscript] TIMEOUT = stuck.
# Run:  $env:PS2X_PAD2='1'
#       .\build_scripts\gfx_tour.ps1 -Name vsall -Script build_scripts\sweeps\vsall.txt -Seconds 3600 -Every 90

wait app=CAppWarning timeout=900
press X hold=10 gap=50 until app!=CAppWarning timeout=3000
label boot
press S hold=10 gap=50 until app=CAppMenuMain timeout=1500
goto boot 6 if app!=CAppMenuMain
poke 0x5E6B30 0xFFF0 w=4
poke 0x5EC2F0 0x7FE w=4
poke 0x5C7942 0
poke 0x5C7943 0
poke 0x5C7944 0
log unlock + 1 round 45s + stage select on
wait ticks=120

label menu
wait app=CAppMenuMain timeout=3000
wait ticks=90
repeat 4
  press U hold=6 gap=6
  press L hold=6 gap=6
end
press D hold=6 gap=30
press R hold=6 gap=30
log menu path D R (expect mode=3)
press X hold=10 gap=60 until app=CAppSelectMain timeout=600
goto vs_ok if mode=3
log NOT VERSUS - backing out, trying pad 2 Start first
press T hold=10 gap=60 until app=CAppMenuMain timeout=300
press S hold=10 gap=60 pad=2
goto menu 2
label vs_ok
log in Versus char select

label charsel
wait ticks=150
press R hold=6 gap=30
press X hold=10 gap=60
log P1 confirmed
press R hold=6 gap=30 pad=2
press X hold=10 gap=90 pad=2
log P2 confirmed
wait ticks=120
press X hold=10 gap=120
log handicap done, stage panel
wait ticks=100
press R hold=6 gap=100
press X hold=10 gap=60
press X hold=10 gap=60 pad=2 until app=CAppFightMain timeout=900
log fight start
wait ticks=300
freeze [0x5AF7F0]+12 250 w=4
freeze [0x5B14C0]+12 250 w=4
fuzz ticks=600 seed=11 max=2 hold=2-8 gap=2-8
fuzz ticks=500 seed=12 max=2 hold=2-8 gap=2-8 pad=2
unfreeze
log fuzz done, letting the fight end
fuzz ticks=2500 seed=13 max=2 hold=2-8 gap=2-8 until app!=CAppFightMain
log left fight
label result
wait ticks=200
press D hold=6 gap=30
press X hold=10 gap=200
goto charsel if app=CAppSelectMain
goto menu if app=CAppMenuMain
goto result 6
log stuck after result - pause routine
press S hold=10 gap=90
press U hold=6 gap=30
press X hold=10 gap=300
goto menu
